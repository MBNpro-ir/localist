if(TARGET flutter_webrtc_plugin)
  get_target_property(audio_sources flutter_webrtc_plugin SOURCES)
  get_target_property(audio_plugin_dir flutter_webrtc_plugin SOURCE_DIR)
  set(audio_dispatch_found FALSE)
  set(audio_base_found FALSE)
  set(audio_screen_found FALSE)
  set(audio_loopback_found FALSE)
  foreach(source IN LISTS audio_sources)
    if(source MATCHES "/application_loopback_capturer.cc$" OR source STREQUAL "application_loopback_capturer.cc")
      get_filename_component(loopback_path "${source}" ABSOLUTE BASE_DIR "${audio_plugin_dir}")
      file(READ "${loopback_path}" loopback)
      string(FIND "${loopback}" "      source_->CaptureFrame(feed.data()," loopback_anchor)
      if(loopback_anchor EQUAL -1)
        message(FATAL_ERROR "WebRTC loopback changed; review source gain/PCM processing")
      endif()
      string(PREPEND loopback "#include \"webrtc_microphone.h\"\n")
      string(REPLACE "      source_->CaptureFrame(feed.data()," "      LocalistProcessLoopbackAudio(feed.data(),sample_rate,out_channels,frames_per_10ms);\n      source_->CaptureFrame(feed.data()," loopback "${loopback}")
      file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/application_loopback_capturer.cc" "${loopback}")
      list(REMOVE_ITEM audio_sources "${source}")
      set(audio_loopback_found TRUE)
    endif()
    if(source MATCHES "/flutter_screen_capture.cc$" OR source STREQUAL "../common/cpp/src/flutter_screen_capture.cc")
      get_filename_component(screen_path "${source}" ABSOLUTE BASE_DIR "${audio_plugin_dir}")
      file(READ "${screen_path}" screen)
      string(FIND "${screen}" "  // VIDEO" screen_anchor)
      if(screen_anchor EQUAL -1)
        message(FATAL_ERROR "WebRTC display capture changed; review audio-only capture")
      endif()
      string(REPLACE "  // VIDEO" "  // Audio-only capture never starts or allocates a desktop video source.\n  const auto video_flag=constraints.find(EncodableValue(\"video\"));\n  if(video_flag!=constraints.end() && TypeIs<bool>(video_flag->second) && !GetValue<bool>(video_flag->second)){\n    params[EncodableValue(\"videoTracks\")]=EncodableValue(EncodableList());\n    base_->local_streams_[uuid]=stream;\n    result->Success(EncodableValue(params));return;\n  }\n  // VIDEO" screen "${screen}")
      file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/flutter_screen_capture.cc" "${screen}")
      list(REMOVE_ITEM audio_sources "${source}")
      set(audio_screen_found TRUE)
    endif()
    if(source MATCHES "/flutter_webrtc_base.cc$" OR source STREQUAL "../common/cpp/src/flutter_webrtc_base.cc")
      get_filename_component(base_path "${source}" ABSOLUTE BASE_DIR "${audio_plugin_dir}")
      file(READ "${base_path}" base)
      string(FIND "${base}" "  audio_processing_ = factory_->GetAudioProcessing();" audio_init)
      if(audio_init EQUAL -1)
        message(FATAL_ERROR "WebRTC audio initialization changed; review silent capture routing")
      endif()
      string(PREPEND base "#include \"webrtc_microphone.h\"\n")
      string(REPLACE "  audio_processing_ = factory_->GetAudioProcessing();"
        "  audio_processing_ = factory_->GetAudioProcessing();\n  std::shared_ptr<EventChannelProxy> pcm_events(EventChannelProxy::Create(messenger_,task_runner_,\"localist/audioFrames\").release());\n  LocalistPrepareCaptureAudio(audio_device_,audio_processing_,[pcm_events](const std::vector<uint8_t>& bytes){pcm_events->Success(EncodableValue(bytes),false);});" base "${base}")
      file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/flutter_webrtc_base.cc" "${base}")
      list(REMOVE_ITEM audio_sources "${source}")
      set(audio_base_found TRUE)
    endif()
    if(source MATCHES "/flutter_webrtc.cc$" OR source STREQUAL "../common/cpp/src/flutter_webrtc.cc")
      get_filename_component(dispatch_path "${source}" ABSOLUTE BASE_DIR "${audio_plugin_dir}")
      file(SHA256 "${dispatch_path}" dispatch_hash)
      if(NOT dispatch_hash STREQUAL "79b17cd9dadca8682d6b302842cc75c47c3a1853b6eac7f99161e5bceb75b697")
        message(FATAL_ERROR "WebRTC dispatch changed; review Localist microphone binding")
      endif()
      file(READ "${dispatch_path}" dispatch)
      string(PREPEND dispatch "#include \"webrtc_microphone.h\"\n")
      string(REPLACE "FlutterWebRTC::~FlutterWebRTC() {}" "FlutterWebRTC::~FlutterWebRTC() { LocalistMicrophonesStop(); }" dispatch "${dispatch}")
      file(READ "${CMAKE_CURRENT_SOURCE_DIR}/microphone_dispatch.inc" microphone_dispatch)
      string(REPLACE "  // Everything below needs the factory." "${microphone_dispatch}\n  // Everything below needs the factory." dispatch "${dispatch}")
      file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/flutter_webrtc.cc" "${dispatch}")
      list(REMOVE_ITEM audio_sources "${source}")
      set(audio_dispatch_found TRUE)
    endif()
  endforeach()
  if(NOT audio_dispatch_found OR NOT audio_base_found OR NOT audio_screen_found OR NOT audio_loopback_found)
