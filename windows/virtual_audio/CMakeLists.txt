if(TARGET flutter_webrtc_plugin)
  get_target_property(audio_sources flutter_webrtc_plugin SOURCES)
  get_target_property(audio_plugin_dir flutter_webrtc_plugin SOURCE_DIR)
  set(audio_dispatch_found FALSE)
  foreach(source IN LISTS audio_sources)
    if(source MATCHES "/flutter_webrtc.cc$" OR source STREQUAL "../common/cpp/src/flutter_webrtc.cc")
      get_filename_component(dispatch_path "${source}" ABSOLUTE BASE_DIR "${audio_plugin_dir}")
      file(SHA256 "${dispatch_path}" dispatch_hash)
      if(NOT dispatch_hash STREQUAL "79b17cd9dadca8682d6b302842cc75c47c3a1853b6eac7f99161e5bceb75b697")
        message(FATAL_ERROR "WebRTC dispatch changed; review Localist microphone binding")
      endif()
      file(READ "${dispatch_path}" dispatch)
      string(PREPEND dispatch "#include \"webrtc_microphone.h\"\n")
      string(REPLACE "FlutterWebRTC::~FlutterWebRTC() {}" "FlutterWebRTC::~FlutterWebRTC() { LocalistMicrophonesStop(); }" dispatch "${dispatch}")
      file(READ "${CMAKE_CURRENT_SOURCE_DIR}/microphone_dispatch.inc" microphone_dispatch)
      string(REPLACE "  // Everything below needs the factory." "${microphone_dispatch}\n  // Everything below needs the factory." dispatch "${dispatch}")
      file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/flutter_webrtc.cc" "${dispatch}")
      list(REMOVE_ITEM audio_sources "${source}")
      set(audio_dispatch_found TRUE)
    endif()
  endforeach()
  if(NOT audio_dispatch_found)
    message(FATAL_ERROR "WebRTC dispatch source was not found")
  endif()
  set_property(TARGET flutter_webrtc_plugin PROPERTY SOURCES "${audio_sources}")
  target_sources(flutter_webrtc_plugin PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/flutter_webrtc.cc" webrtc_microphone.cpp cable_output.cpp)
  target_link_libraries(flutter_webrtc_plugin PRIVATE avrt ole32 uuid)
  target_include_directories(flutter_webrtc_plugin PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  if(LOCALIST_AUDIO_QA_PIPE)
    if(NOT LOCALIST_CAPTURE_QA_RUNNER)
      message(FATAL_ERROR "QA pipes require an isolated QA runner")
    endif()
    target_compile_definitions(flutter_webrtc_plugin PRIVATE LOCALIST_AUDIO_QA_PIPE=1)
  endif()
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/runtime/localistaudio.cat")
    install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/runtime/" DESTINATION "${CMAKE_INSTALL_PREFIX}/drivers/LocalistAudio" COMPONENT Runtime)
  endif()
  install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/virtual_audio/upstream/LICENSE"
    DESTINATION "${CMAKE_INSTALL_PREFIX}/licenses" RENAME "LocalistAudio-MIT.txt" COMPONENT Runtime)
  install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/virtual_audio/MICROSOFT-SAMPLE-LICENSE"
    DESTINATION "${CMAKE_INSTALL_PREFIX}/licenses" RENAME "LocalistAudio-MS-PL.txt" COMPONENT Runtime)
endif()
